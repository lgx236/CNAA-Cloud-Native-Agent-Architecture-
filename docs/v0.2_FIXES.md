### 2️⃣ 添加 SQLite 持久化存储 (替换 in-memory)

**目标**: 只要一个文件就能存所有数据

创建文件 `cloud/storage/sqlite_memory_store.py`:

```python
"""Simple SQLite storage backend."""
import sqlite3
import json
from typing import Any
from cloud.storage.memory_store import MemoryInterface
from cnaa.models import Memory, MemorySummary, MemoryType
from datetime import datetime


class SQLiteMemoryStore(MemoryInterface):
    """Minimal SQLite storage - one file per agent data."""
    
    def __init__(self, db_path: str = "cnaa_memories.db"):
        self.db_path = db_path
        self._connect()
        self._create_tables()
    
    def _connect(self):
        """Connect to database."""
        self.conn = sqlite3.connect(self.db_path)
        self.conn.row_factory = sqlite3.Row
    
    def _create_tables(self):
        """Create tables if not exist."""
        cursor = self.conn.cursor()
        
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS memories (
                memory_id TEXT PRIMARY KEY,
                agent_id TEXT NOT NULL,
                type TEXT NOT NULL,
                content TEXT NOT NULL,
                tags TEXT DEFAULT '[]',
                completion_score REAL DEFAULT 0.0,
                timestamp TEXT NOT NULL
            )
        """)
        
        cursor.execute("CREATE INDEX IF NOT EXISTS idx_agent ON memories(agent_id)")
        cursor.execute("CREATE INDEX IF NOT EXISTS idx_timestamp ON memories(timestamp)")
        
        self.conn.commit()
    
    def store_memory(self, memory: Memory) -> dict[str, Any]:
        """Save memory to SQLite."""
        cursor = self.conn.cursor()
        
        try:
            cursor.execute("""
                INSERT OR REPLACE INTO memories 
                (memory_id, agent_id, type, content, tags, completion_score, timestamp)
                VALUES (?, ?, ?, ?, ?, ?, ?)
            """, (
                memory.memory_id,
                memory.agent_id,
                memory.type.value,
                json.dumps(memory.content),
                json.dumps(memory.tags),
                memory.completion_score,
                memory.timestamp.isoformat(),
            ))
            
            self.conn.commit()
            return {"status": "ok", "memory_id": memory.memory_id}
            
        except Exception as e:
            self.conn.rollback()
            return {"status": "error", "message": str(e)}
    
    def get_memory(self, agent_id: str, memory_id: str) -> Memory | None:
        """Load single memory."""
        cursor = self.conn.cursor()
        cursor.execute("""
            SELECT * FROM memories 
            WHERE agent_id = ? AND memory_id = ?
        """, (agent_id, memory_id))
        
        row = cursor.fetchone()
        if not row:
            return None
        
        return self._row_to_memory(dict(row))
    
    def list_memories(
        self,
        agent_id: str,
        memory_type: MemoryType | None = None,
        tags: list[str] | None = None,
        auth_context: dict[str, Any] | None = None,
        start_time: datetime | None = None,
        end_time: datetime | None = None,
        limit: int | None = None,
        reverse: bool = False,
    ) -> list[MemorySummary]:
        """List memories for agent."""
        cursor = self.conn.cursor()
        
        query = "SELECT * FROM memories WHERE agent_id = ?"
        params = [agent_id]
        
        if memory_type:
            query += " AND type = ?"
            params.append(memory_type.value)
        
        if tags:
            placeholders = ",".join(["?" for _ in tags])
            query += f" AND tags LIKE '%{placeholders}%'"
            params.extend(tags)
        
        query += " ORDER BY timestamp DESC"
        if limit:
            query += f" LIMIT {limit}"
        
        cursor.execute(query, params)
        
        return [self._row_to_summary(dict(row)) for row in cursor.fetchall()]
    
    def delete_memory(self, agent_id: str, memory_id: str) -> dict[str, Any]:
        """Delete a memory."""
        cursor = self.conn.cursor()
        cursor.execute(
            "DELETE FROM memories WHERE agent_id = ? AND memory_id = ?",
            (agent_id, memory_id)
        )
        self.conn.commit()
        
        return {"status": "ok"}
    
    def _row_to_memory(self, row: dict) -> Memory:
        """Convert DB row to Memory object."""
        return Memory(
            memory_id=row["memory_id"],
            agent_id=row["agent_id"],
            type=MemoryType(row["type"]),
            content=json.loads(row["content"]),
            tags=json.loads(row["tags"]),
            completion_score=row["completion_score"] or 0.0,
            timestamp=datetime.fromisoformat(row["timestamp"]),
        )
    
    def _row_to_summary(self, row: dict) -> MemorySummary:
        """Convert DB row to MemorySummary."""
        return MemorySummary(
            memory_id=row["memory_id"],
            tags=json.loads(row["tags"]),
            completion_score=row["completion_score"] or 0.0,
            timestamp=datetime.fromisoformat(row["timestamp"]),
        )
    
    def tag_short_term(self, agent_id: str, tags: list[str]) -> dict[str, Any]:
        return {"status": "ok"}
```

### 3️⃣ 添加简单备份脚本

创建文件 `scripts/backup.sh`:

```bash
#!/bin/bash
# Simple daily backup script

BACKUP_DIR="./backups"
DATE=$(date +%Y%m%d_%H%M%S)
mkdir -p $BACKUP_DIR

# Backup all databases
cp *.db $BACKUP_DIR/
cp *.sqlite $BACKUP_DIR/

# Compress old backups (keep last 7 days)
find $BACKUP_DIR -name "*.db.gz" -mtime +7 -delete

echo "Backup created: $BACKUP_DIR/cnaa_$DATE.db"
```

给执行权限:
```bash
chmod +x scripts/backup.sh
crontab -e  # 添加：0 2 * * * /path/to/scripts/backup.sh
```

### 5️⃣ 性能压力测试脚本

创建文件 `tests/stress_test.py`:

```python
"""Simple stress test script."""
import time
import requests
import concurrent.futures
from random import randint

BASE_URL = "http://localhost:8080"
NUM_REQUESTS = 1000
CONCURRENT_USERS = 10

def send_request(i):
    """Send single request."""
    payload = {
        "jsonrpc": "2.0",
        "id": i,
        "method": "tools/call",
        "params": {
            "name": "cnaa_store_memory",
            "arguments": {
                "agent_id": f"stress-agent-{i % 10}",
                "memory_id": f"mem-{i}",
                "type": "long_term",
                "content": {"task": f"Test task {i}"},
            }
        }
    }
    
    start = time.time()
    resp = requests.post(f"{BASE_URL}/mcp", json=payload, timeout=5)
    elapsed = time.time() - start
    
    return resp.status_code == 200, elapsed

def run_stress_test():
    """Run multi-user stress test."""
    print(f"\n🚀 Stress Test: {NUM_REQUESTS} requests, {CONCURRENT_USERS} concurrent")
    
    with concurrent.futures.ThreadPoolExecutor(max_workers=CONCURRENT_USERS) as executor:
        futures = [executor.submit(send_request, i) for i in range(NUM_REQUESTS)]
        results = [f.result() for f in concurrent.futures.as_completed(futures)]
    
    success_count = sum(1 for s, _ in results if s)
    total_time = sum(t for _, t in results)
    
    print(f"\n✅ Results:")
    print(f"   Successes: {success_count}/{NUM_REQUESTS}")
    print(f"   Total time: {total_time:.2f}s")
    print(f"   Throughput: {NUM_REQUESTS/total_time:.1f} req/s")
    print(f"   Avg latency: {total_time/NUM_REQUESTS*1000:.1f}ms")

if __name__ == "__main__":
    run_stress_test()
```

运行方法:
```bash
# 先启动 server
python server.py --port 8080 &

# 然后跑压力测试
python tests/stress_test.py
```

1. 改一行代码启用认证

- ❌ 不搞复杂的 RBAC
